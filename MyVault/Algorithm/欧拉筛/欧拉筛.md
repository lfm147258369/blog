#include <bits/stdc++.h>  
using namespace std;  
const int maxn = 1e7+10;  
bitset<maxn>pri;  
int primes[maxn], pp = 0;  
  
int main()  
{  
    int N = 1e7;  
    int cnt = 0;  
    //欧拉筛  
    for (int i = 2; i <= N; ++i)  
    {  
        if (!pri[i])  
        {  
            primes[++pp] = i;  
            cnt++;  
        }  
        for (int j = 1; primes[j] * i <= N; ++ j)  
        {  
            pri[primes[j] * i]  = 1;  
            if (i % primes[j] == 0)  
            {  
                break;  
            }  
        }  
    }  
    printf("%d", cnt);  
    return 0;  
}
