

### 重述问题
将字符串 str1 转换为 str2 所需的最少操作次数，允许的操作包括插入、删除和修改字符。

### 解题思路
1. **最后一步分析**：考虑将 str1 的前 i 个字符转换为 str2 的前 j 个字符的最后一步操作，可能为插入、删除或修改。
2. **子问题划分**：根据最后一步操作，问题可分解为处理更短长度的子问题。
3. **边界条件**：当任一字符串长度为0时，操作次数为另一字符串的长度。
4. **状态转移方程**：
   - 如果字符相同：`dp[i][j] = dp[i-1][j-1]`
   - 否则：`dp[i][j] = min(插入, 删除, 修改) + 1`

### 暴力递归（超时，用于理解思路）
```cpp
class Solution 

public:
    int editDistance(string str1, string str2) 
    {
        return dfs(str1.size(), str2.size(), str1, str2);
    }
    
private:
    int dfs(int i, int j, string& s1, string& s2)
    {
        if (i == 0) return j; // str1为空，插入j次
        if (j == 0) return i; // str2为空，删除i次
        if (s1[i-1] == s2[j-1]) 
        {
            return dfs(i-1, j-1, s1, s2); // 字符相同，无需操作
        }
        // 三种操作取最小值+1
        int del = dfs(i-1, j, s1, s2);     // 删除s1[i]
        int ins = dfs(i, j-1, s1, s2);     // 插入s2[j]
        int rep = dfs(i-1, j-1, s1, s2);   // 替换s1[i]为s2[j]
        return 1 + min({del, ins, rep});
    }
};
```

### 记忆化搜索（优化递归）
```cpp
class Solution
{
public:
    vector<vector&lt;int&gt;> memo; // 记忆数组
    
    int editDistance(string str1, string str2)
    {
        int m = str1.size(), n = str2.size();
        memo.assign(m+1, vector&lt;int&gt;(n+1, -1));
        return dfs(m, n, str1, str2);
    }
    
    int dfs(int i, int j, string& s1, string& s2)
    {
        if (i == 0) return j;
        if (j == 0) return i;
        if (memo[i][j] != -1) return memo[i][j]; // 已计算过
        
        if (s1[i-1] == s2[j-1])
        {
            memo[i][j] = dfs(i-1, j-1, s1, s2);
        } 
        else 
        {
            int del = dfs(i-1, j, s1, s2);
            int ins = dfs(i, j-1, s1, s2);
            int rep = dfs(i-1, j-1, s1, s2);
            memo[i][j] = 1 + min({del, ins, rep});
        }
        return memo[i][j];
    }
};
```

### 动态规划（二维数组）
```cpp
class Solution 
{
public:
    int editDistance(string str1, string str2) 
    {
        int m = str1.size(), n = str2.size();
        vector<vector&lt;int&gt;> dp(m+1, vector&lt;int&gt;(n+1, 0));
        
        // 初始化边界条件
        for (int i = 0; i <= m; ++i) dp[i][0] = i;
        for (int j = 0; j <= n; ++j) dp[0][j] = j;
        
        for (int i = 1; i <= m; ++i) 
        {
            for (int j = 1; j <= n; ++j) 
            {
                if (str1[i-1] == str2[j-1]) 
                {
                    dp[i][j] = dp[i-1][j-1];
                } else 
                {
                    dp[i][j] = 1 + min({
                        dp[i-1][j],   // 删除
                        dp[i][j-1],   // 插入
                        dp[i-1][j-1]  // 替换
                    });
                }
            }
        }
        return dp[m][n];
    }
};
```

### 动态规划（滚动数组优化）
```cpp
class Solution
{
public:
    int editDistance(string str1, string str2)
    {
        int m = str1.size(), n = str2.size();
        vector&lt;int&gt; prev(n+1, 0), curr(n+1, 0);
        
        // 初始化prev为str1长度为0时的操作次数
        for (int j = 0; j <= n; ++j) prev[j] = j;
        
        for (int i = 1; i <= m; ++i)
        {
            curr[0] = i; // 初始化当前行首列
            for (int j = 1; j <= n; ++j)
            {
                if (str1[i-1] == str2[j-1])
                {
                    curr[j] = prev[j-1];
                } 
                else
                {
                    curr[j] = 1 + min({
                        prev[j],    // 删除
                        curr[j-1], // 插入
                        prev[j-1]  // 替换
                    });
                }
            }
            swap(prev, curr); // 滚动数组
        }
        return prev[n];
    }
};
```

### 动态规划（单数组优化）
```cpp
class Solution {
public:
    int editDistance(string str1, string str2)
    {
        int m = str1.size(), n = str2.size();
        vector&lt;int&gt; dp(n+1, 0);
        
        // 初始化：当str1长度为0时，操作次数为str2的长度
        for (int j = 0; j <= n; ++j) dp[j] = j;
        
        for (int i = 1; i <= m; ++i)
        {
            int prev_diag = dp[0]; // 保存左上角的值
            dp[0] = i; // 当前行首列初始化为i
            
            for (int j = 1; j <= n; ++j)
            {
                int temp = dp[j]; // 保存旧值作为下一轮的prev_diag
                if (str1[i-1] == str2[j-1])
                {
                    dp[j] = prev_diag;
                }
                else 
                {
                    dp[j] = 1 + min({
                        dp[j-1],   // 插入（左侧）
                        dp[j],     // 删除（上方）
                        prev_diag  // 替换（左上方）
                    });
                }
                prev_diag = temp; // 更新左上角
            }
        }
        return dp[n];
    }
};
```

### 总结
所有解法均基于动态规划思想，通过逐步优化空间复杂度：
1. **暴力递归**：直接递归但存在重复计算。
2. **记忆化搜索**：通过缓存子问题结果优化递归。
3. **二维DP**：显式构建二维状态表。
4. **滚动数组**：利用两行数组减少空间占用。
5. **单数组优化**：仅用一维数组实现，空间复杂度降至 O(n)。
