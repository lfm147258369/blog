字母之二# 思路
- 先把每个子数组存入临时变量
- 修改临时子数组：判断字母，小写转大写， 大写转小写
- 当层数到达长度，追加近答案数组

		ans.emplace_back(tmp);

	//选
	
		tmp[x] = s[x];
		
		dfs(x + 1, s);
		
		tmp.pop_back();
		
		//不选
		
		dfs(x + 1, s);
	
	}
	
	vector&lt;string&gt; letterCasePermutation(string s)
	
	{
	
	int n = s.size();
	
	tmp.resize(n);
	
	dfs(0, s);
	
	return ans;
	
}

};

class Solution
{
public:
    int n;
    string tmp;
    vector&lt;string&gt; ans;
    void dfs(int x, string & s)
    {
        if (n == x)
        {
            for (int i = 0; i < tmp.size(); i++)
            {
                if (tmp[i] >= 'A' && tmp[i] <= 'Z')
                {
                    tmp[i] += 32;
                }
                else if (tmp[i] >= 'a' && tmp[i] <= 'z')
                {
                    tmp[i] -= 32;
                }
            }
            ans.emplace_back(tmp);
            return ;
        }

        tmp[x] = s[x];
        dfs(x + 1, s);
        tmp[x] = ' ';

    }
    vector&lt;string&gt; letterCasePermutation(string s)
    {
        n = s.size();
        tmp.resize(n);
        dfs(0, s);
        return ans;
    }
};


int main()
{
    Solution s1;
    string s = "a1b2";
    vector&lt;string&gt; res = s1.letterCasePermutation(s);
    for (auto & t : res)
    {
        cout << t << endl;
    }
    return 0;
}

- 分支情况应该是：
- 有n个字母，分为改其中0个，1个 ......n个。
- 那分支就是改和不改

```cpp
class Solution
{
public:
    int n;
    string tmp;
    vector&lt;string&gt; ans;
    void dfs(int x, string & s)
    {
        if (n == x)
        {
            ans.emplace_back(tmp);
            return ;
        }
        //不改
        tmp[x] = s[x];
        dfs(x + 1, s);

        //改
        if  (((tmp[x] >= 'A') && (tmp[x] <= 'Z') )   ||  ((tmp[x] >= 'a') && (tmp[x] <= 'z')) )
        {
            if ( (tmp[x] >= 'A') && (tmp[x] <= 'Z') )
            {
                tmp[x] += 32;
            }
            else if ((tmp[x] >= 'a') && (tmp[x] <= 'z'))
            {
                tmp[x] -= 32;
            }
            dfs(x + 1, s);
        }
    }
    vector&lt;string&gt; letterCasePermutation(string s)
    {
        n = s.size();
        tmp.resize(n);
        dfs(0, s);
        return ans;
    }
};

```


- 如此找到两个正确的分支，改和不改，分支下去
- 不用考虑组合合法，因为不是选和不选，不会出现空的
- 注意在改的分支那里，因为它是两个分支，dfs要放在if里面
- 否则就会出现重复的组合
