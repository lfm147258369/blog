```cpp
#include &lt;iostream&gt;
using namespace std;
//类模板成员函数类外实现

template<class T1, class T2>
class Person
{
public:
    Person(T1 name, T2 age);
    // {
    //     this->m_Name = name;
    //     this->m_Age = age;
    // }
    void showPerson();
    // {
    //     cout << "姓名 " << this->m_Name << "年龄 " << this->m_Age;
    // }
    T1 m_Name;
    T2 m_Age;
};
template<class T1, class T2>
Person<T1, T2>::Person(T1 name, T2 age)
{
    this->m_Name = name;
    this->m_Age = age;
}

//成员函数的类外实现
template <class T1, class T2>
void Person<T1, T2>::showPerson()
{
    cout << "姓名 " << this->m_Name << "年龄 " << this->m_Age;
}


void test01()
{
    Person<string, int> p("white", 52);
    p.showPerson();
}
int main()
{
    test01();
    return 0;
}

```


- 需要注意在类外的实现要加上模板参数列表
- 由于参数是类模板的类型，在类外实现还要再加上template
```cpp
//成员函数的类外实现
template <class T1, class T2>
void Person<T1, T2>::showPerson()
{
    cout << "姓名 " << this->m_Name << "年龄 " << this->m_Age;
}
```


- 在类外实现的时候，还要注意加上类模板参数列表，既然加上了类模板参数列表就少不了template
```cpp
//成员函数的类外实现
template <class T1, class T2>
void Person<T1, T2>::showPerson()
{
    cout << "姓名 " << this->m_Name << "年龄 " << this->m_Age;
}
```
