 - 类内实现和类外实现

- 类内实现直接在类内实现一个有元函数即可使用全局函数来访问类中的私有成员属性

- 类外实现要让编译器提前知道类是模板类,函数也是模板,所以要把类的实现还有函数的实现放在开头


先看类内实现
```cpp
#include <iostream>  
using namespace std;  
void printPerson2(Person<T1, T2> p)  
{  
    cout << "姓名: " << p.m_Name << " " << "年龄" << p.m_Age << endl;  
}  
  
template<class T1, class T2>  
class Person  
{  
    //全局函数类内实现  
    friend void printPerson(Person <T1, T2> p)  
    {         cout << "姓名: " << p.m_Name << " " << "年龄" << p.m_Age << endl;  
    }
public:  
    Person(T1 name, T2 age)  
    {  
        this->m_Name = name;  
        this->m_Age = age;  
    }  
private:  
    T1 m_Name;  
    T2 m_Age;  
};  
  
//1.  
void test01()  
{  
    Person<string, int>p("TOm", 1000);  
    // printPerson(p);  
}  
  
int main()  
{  
    test01();  
    return 0;  
}
```


- 再看类外实现


```cpp
#include <iostream>  
using namespace std;  

//提前让编译器知道Person类存在  
template<class T1, class T2>  
class Person;  

//提前让编译器知道这个函数是模板函数
template<class T1, class T2>  
void printPerson2(Person<T1, T2> p)  
{  
    cout << "姓名: " << p.m_Name << " " << "年龄" << p.m_Age << endl;  
}  
  
template<class T1, class T2>  
class Person  
{  
    //全局函数类外实现  
    // 加空参数列表  
    //如果全局函数是类外实现需要让编译器提前知道函数存在  
    friend void printPerson2<>(Person <T1, T2> p);  
public:  
    Person(T1 name, T2 age)  
    {  
        this->m_Name = name;  
        this->m_Age = age;  
    }  
private:  
    T1 m_Name;  
    T2 m_Age;  
};  
  
//2.类外实现  
void test02()  
{  
    Person<string, int> p("Jerry", 20);  
    printPerson2(p);  
}  
  
int main()  
{  
    test02();  
    return 0;  
}
```
