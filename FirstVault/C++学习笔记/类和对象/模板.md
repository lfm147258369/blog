
# 模板的基本语法


- 1.自动类型推到
让编译器自动推导数据的类型

```cpp
template<typename T>//声明一个模板 告诉编译器紧跟着的T不要报错，T是一个通用数据类型
void mySwap(T &a, T &b)
{
    T tmp = a;
    a = b;
    b = tmp;
}
mySwap(a, b);
cout << "a = " << a << endl;
cout << "b = " << b << endl;

```

- 2.显示指定类型
就是让编译器不用自己判断了，自己提前告诉它

```cpp
template<typename T>//声明一个模板 告诉编译器紧跟着的T不要报错，T是一个通用数据类型
void mySwap(T &a, T &b)
{
    T tmp = a;
    a = b;
    b = tmp;
}
mySwap<int>(a, b);
cout << "a = " << a << endl;
cout << "b = " << b << endl;

```

- 这样写就不需要因为数据类型不同写不一样的交换函数了，因为底层都是两个变量交换，区别只是类型


# 模板的注意事项

- 1.自动推导类型必须要推导出一致的数据类型T才能使用
- c为char类型，a为int类型，编译器无法推导
```cpp
template<typename T>//typename可以替换为class关键字
void mySoty(T &a, T &b)
{
    T tmp = a;
    a = b;
    b = tmp;
}

//1.自动类型推到：必须要推到出一致的数据类型T才能使用
void test01()
{
    int a = 10;
    int b = 20;
    char c = 'c';
    //mySoty(a, b);//正确

    //mySwap(a, c);//推导不出一致的T类型，编译错误

    cout << "a = " << a << " b = " << b << endl;

}
```

- 2.模板要指定出数据类型，比如不能用auto，无参的时候要随便指定一个类型 

```cpp
//2,模板必须要确定出T的数据类型，不能使用auto关键字
template<class T>
void func()
{
    cout << "func 调用" << endl;
}
void test02()
{
    func<int>();//随便给它一个int类型参数，编译通过
}
```
 - 这里func函数没有参数，编译器也是无法推导的。需要自己来指定

# 数组排序

- 我们分别用两种数组来测试我们的模板函数，是不是可以排序不同类型的数组
- 我们使用选择排序的算法
```cpp
//排序算法
template<class T>
void MySwap(T &a, T &b)
{
    T tmp = a;
    a = b;
    b = tmp;
}
template<class T>
void MySort(T arr[], int len)
{
    for (int i = 0; i < len - 1; i++)
    {
        int max = i;//认定的最大值下标
        for (int j = i + 1; j < len; j++)
        {
            //认定的最大值下标比遍历出的元素小，则更新认定的最大值下标
            if (arr[max] < arr[j])
            {
                max = j;
            }
        }
        if (i != max)
        {
            MySwap(arr[i], arr[max]);
        }
    }
}
```
- 我们让编译来自动识别数组的类型
- 然后我们创建一个字符数组和一个整型数组来确认它们是否都能被正确的排序
- 
```cpp

void test01()  
{  
    //测试char数组  
    char charArr[] = "badcfe";  
    int num = sizeof(charArr) / sizeof(charArr[0]);  
    MySort(charArr, num);  
    PrintArray(charArr, num);  
}
```

```cpp
void test02()
{
    //测试int数组
    int intArr[] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5};
    int num = sizeof(intArr) / sizeof(intArr[0]);
    MySort(intArr, num);
    PrintArray(intArr, num);
}

```
- 结果是二者都可以排序


# 普通函数和函数模板的区别
```cpp
#include <iostream>
using namespace std;

//普通函数和函数模板的区别


//1.普通函数可以发生隐式类型转换
//2.函数模板 用自动类型推导，不可以发生隐式类型转换

//普通函数
int myAdd01(int a, int b)
{
    return a + b;
}

//函数模板
template<typename T>
T myAdd02(T a, T b)
{
    return a + b;
}

void test01()
{
    int a = 10;
    int b = 20;
    //a - 97, c - 99
    char c = 'c';//把字符型变量c转换为int型变量

    cout << myAdd01(a, c) << endl; //a = 10 + c= 99 = 109
    
    //自动类型推导 不会发生隐式类型转换
    //cout << myAdd02(a, c) << endl; //a = 10 + c= 99 = 109

    //显示指定类型 可以发生隐式类型转换
    cout << myAdd02<int>(a, c) << endl;
}

```
- 我们写了两个函数，首先来看我们写的普通函数
```cpp

//普通函数
int myAdd01(int a, int b)
{
    return a + b;
}
```

- 如果我们传入一个char类型的数据，他会将数据类型转化为int类型
- 那么在函数模板中呢？
```cpp
//函数模板
template<typename T>
T myAdd02(T a, T b)
{
    return a + b;
}
void test01()
{
    int a = 10;
    int b = 20;
    //a - 97, c - 99
    char c = 'c';//把字符型变量c转换为int型变量

    cout << myAdd01(a, c) << endl; //a = 10 + c= 99 = 109

    //自动类型推导 不会发生隐式类型转换
    //cout << myAdd02(a, c) << endl; //a = 10 + c= 99 = 109

    //显示指定类型 可以发生隐式类型转换
    cout << myAdd02<int>(a, c) << endl;
}

```
- 当我们使用自动类型推导的时候，我们可以发现，IDE报错了
- 因为它无法帮我们进行隐式类型转换
- 当我们指定类型的时候，他会进行隐式类型转换，将我们传入的char类型c转化为int


# 函数模板和普通函数的调用规则
- 1.如果模板函数和普通函数都可以调用，那么优先调用普通函数

```cpp
void myPrint(int a, int b)
{
    cout << "调用普通的函数" << endl;
}

template<class T>
void myPrint(T a, T b)
{
    cout << "调用的模板" << endl;
}

void test01()
{
    int a = 10;
    int b = 20;
    myPrint(c1, c2);
}

```
- 上述的代码发生了函数重载，当我们传入了两个整型变量a和b的时候，普通函数和模板函数都可以调用，根据第一条规则，它会优先调用普通的函数
- 故输出的结果为
```
调用普通的函数
```

- 2.可以通过空模板参数列表强制调用模板函数
```cpp
void myPrint(int a, int b);
//{
//    cout << "调用普通的函数" << endl;
//}

template<class T>
void myPrint(T a, T b)
{
    cout << "调用的模板" << endl;
}

void test01()
{
    int a = 10;
    int b = 20;


    //通过空函数的参数列表 强制调用函数模板
    myPrint<>(a, b);
}
```
- 我们把普通函数注释掉，将其变为函数的声明，然后我们相要强行调用模板函数可以通过空模板参数列表来强制调用


- 3.函数模板也可以发生函数重载
```cpp
template<class T>  
void myPrint(T a, T b)  
{  
    cout << "调用的模板" << endl;  
}  
  
template<class T>  
void myPrint(T a, T b, T c)  
{  
    cout << "调用重载的模板" << endl;  
}

void test01()
{
    int a = 10;
    int b = 20;


    myPrint(a, b, 20);

}

```
- 发生了参数不同的重载


- 4.会默认调用最合适的
```cpp
void myPrint(int a, int b)
{
    cout << "调用普通的函数" << endl;
}

template<class T>
void myPrint(T a, T b)
{
    cout << "调用的模板" << endl;
}

template<class T>
void myPrint(T a, T b, T c)
{
    cout << "调用重载的模板" << endl;
}

void test01()
{
    int a = 10;
    int b = 20;

    char c1 = 'a';
    char c2 = 'b';

    myPrint(c1, c2);
}

```


- myPrint函数同名的有多个，c1和c2是char类型，这里最适合的是模板函数，不用再发生隐式类型转换

#  模板的具体化

- 我们在使用模板的时候，当我们传入自定义类型的时候，需要进行具体化，否则编译器无法进行自动推导

```cpp
#include <iostream>
#include <string>
using namespace std;

//模板的局限性

//对比两个数据是否相等
class Person
{
public:
    Person(string name, int age)
    {
        this->m_Name = name;
        this->m_Age = age;
    }
public:
    string m_Name;
    int m_Age;
};
template <class T>
bool mycompare(T & a, T & b)
{
    if (a == b)
    {
        return true;
    }
    else
    {
        return false;
    }
}
//利用具体化的Person的版本实现代码，具体化优先调用

template<>bool mycompare(Person &a, Person &b)
{
    if (a.m_Name == b.m_Name && a.m_Age == a.m_Age)
    {
        return true;
    }
    else
    {
        return false;
    }
}
void test01()
{
    int a = 10;
    int b = 20;

    bool ret = mycompare(a, b);
    if (ret)
        cout << "==" << endl;
    else
        cout << "no" << endl;
}
void test02()
{
    Person p1("Tom", 10);
    Person p2("Tom", 10);
    bool ret = mycompare(p1, p2);
    if (ret)
        cout << "相同" << endl;
    else
        cout << "不相同" << endl;
}
int main()
{
//    test01();
    test02();
    return 0;
}


```

- 我们先写了一个模板函数，作用是比较两个变量
```cpp
bool mycompare(T & a, T & b)
{
    if (a == b)
    {
        return true;
    }
    else
    {
        return false;
    }
}

{
    int a = 10;
    int b = 20;

    bool ret = mycompare(a, b);
    if (ret)
        cout << "==" << endl;
    else

}

```

当我们传入的是自定义类型的时候


# 类模板


```cpp
#include <iostream>
#include <string>
using namespace std;

//类模板
template<class NameType, class AgeType>
class Person
{
public:
    Person(NameType name, AgeType age)
    {
        this->m_Name = name;
        this->m_Age = age;
    }
    void Show()
    {
        cout << "name = " << m_Name << "  "  << "age = " << this->m_Age;
    }
    NameType m_Name;
    AgeType m_Age;
};

void test01()
{
    //模板参数列表
    Person<string, int>  p1("zhangsan", 99);
    p1.Show();

}

int main()
{
    test01();
    return 0;
}

```
- 我们将类模板作为参数

```cpp

//类模板
template<class NameType, class AgeType>
class Person
{
public:
    Person(NameType name, AgeType age)
    {
        this->m_Name = name;
        this->m_Age = age;
    }
    void Show()
    {
        cout << "name = " << m_Name << "  "  << "age = " << this->m_Age;
    }
    NameType m_Name;
    AgeType m_Age;
};
```
- 这样可以让类模板作为有参构造函数的参数，也可以用来创建成员变量


# 类模板和函数模板区别


 - 类的定义
```cpp

#include <iostream>
#include <string>

template<class NameType, class AgeType = int>//默认类型

{
public:
    Person(NameType name, AgeType age)
    {
        this->m_Name = name;
        this->m_Age = age;

    {

```
- 类模板没有自动类型推导， 只能指定类型
