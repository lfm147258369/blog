# 代码

```cpp
#include <bits/stdc++.h>  
using namespace std;  
  
//类模板中成员函数的创建时机  
//类模板中成员函数在调用时才创建  
  
class Person1  
{  
public:  
    void showPerson1()  
    {  
        cout << "Person1 show" << endl;  
    }  
};  
  
  
class Person2  
{  
public:  
    void showPerson2()  
    {  
        cout << "Person2 show" << endl;  
    }  
};  
  
template<class T>  
class MyClass  
{  
public:  
  
    T obj;  
  
    //类模板中的成员函数  
    void func1()  
    {  
        obj.showPerson1();  
    }  
  
    void func2()  
    {  
        obj.showPerson2();  
    }  
};  
  
  
void test01()  
{  
    MyClass<Person2> m;  
    //m.func1();  
    m.func2();  
}  
  
int main()  
{  
    test01();  
    return 0;  
}
```


- 注意的点：类模板的成员函数不是一开始就创建而是调用时创建
- 是我们在这里调用然后创建了对应的成员函数
- 还要注意你制定的类型要对应你调用的类型，如果是`MyClass<Person1> m` 那么就无法调用`m.func2()` 了。
- 
```cpp
void test01()  
{  
    MyClass<Person2> m;  
    //m.func1();  
    m.func2();  
}  
```
